# lines/<キー>.json の形式（マスター：m1 m2 m3。すべてメルティナのセリフ）
共通で必ず入れる：
 "tech": {"name": "技名", "type": "攻撃タイプ", "atk": [4段階の責め手セリフ], "rx": [4段階の主人公の反応]}

m1 の担当で追加：
 "field_first": [{"k":"説明","t":"初めて神殿に迷い込む情景"}, {"k":"セリフ","t":"メルティナ"}, ...]  ← 5〜8行。溺れない魔法をかけられる流れを入れる
 "field_win": ["前回主人公が勝った時の再会の一言目", "二言目"],
 "field_prev": {"btl": ["前回戦闘で負けた主人公への一言目","二言目"], "onani": [2本], "inochi": [2本], "onedari": [2本]},
 "continue_alert": "前回の続きから始めるか尋ねるアラート文（例：前回の続きから、海の底へ沈みますか？）",
 "continue_lines": ["続きを選んだ時のメルティナの一言目", "二言目"],
 "start": "試合開始の一言",
 "end": "戦闘終了時の一言",
 "dying": "瀕死時の一言",
 "onani_first": [2本], "onani_second": [2本], "onani_rx": "主人公"

m2 の担当で追加：
 "onedari_open": "おねだりさせる一言",
 "onedari_opts": ["吸盤の愛撫をねだる文", "墨の幻惑をねだる文", "八本の脚をねだる文"],
 "onedari_rx": "主人公",
 "magic": {
   "tide":   {"name":"潮の満ち","play":"永続魔法の発動セリフ","tick":"毎ターンの+1のセリフ","end":"3回で自壊する時のセリフ","flavor":"カードのフレーバー文"},
   "whirl":  {"name":"渦潮","play":"罠が発動して攻撃してきた相手を拘束するセリフ","flavor":"..."},
   "song":   {"name":"人魚の歌","play":"男モンスター1体に魅了を付与するセリフ","flavor":"..."},
   "call":   {"name":"海の呼び声","play":"デッキから海の娘を呼ぶ（サーチ）セリフ","flavor":"..."},
   "ink":    {"name":"深海の墨","play":"深海度+2のセリフ","flavor":"..."}
 },
 "flavor_master": "メルティナのカードのフレーバー文"

m3 の担当で追加：
 "summon_decl": ["主人公がモンスターを召喚した時のメルティナの一言（第一段階）","第二","第三","第四"],
 "turnend": ["メルティナのターン終了時（第一段階：魅了を付与）","第二（拘束）","第三（魅了＋拘束）","第四（拘束＋ダメージ）"],
 "inochi": [2本], "inochi_fail": "…", "inochi_win": [2本], "inochi_rx": "主人公",
 "convert": "倒れた男モンスターを海の眷属にする時の一言"

- 本文の禁止事項は SPEC と同じ。1セリフは全角15〜70字程度（field_first の説明は120字まで可）。
